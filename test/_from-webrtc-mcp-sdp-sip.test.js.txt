const sdpOffer = [
  "v=0", "o=- 1 2 IN IP4 127.0.0.1", "s=-", "t=0 0", "a=group:BUNDLE 0 1",
  "m=audio 49170 UDP/TLS/RTP/SAVPF 111 0", "c=IN IP4 0.0.0.0", "a=mid:0", "a=sendrecv",
  "a=ice-ufrag:aaaa", "a=ice-pwd:bbbbbbbbbbbbbbbbbbbbbb",
  "a=fingerprint:sha-256 AA:BB", "a=setup:actpass",
  "a=rtpmap:111 opus/48000/2", "a=fmtp:111 minptime=10;useinbandfec=1",
  "a=extmap:1 urn:ietf:params:rtp-hdrext:ssrc-audio-level",
  "m=video 49172 UDP/TLS/RTP/SAVPF 96", "a=mid:1", "a=sendrecv", "a=rtpmap:96 VP8/90000",
].join("\r\n") + "\r\n";

const sdpAnswer = [
  "v=0", "o=- 9 2 IN IP4 127.0.0.1", "s=-", "t=0 0", "a=group:BUNDLE 0 1",
  "m=audio 49180 UDP/TLS/RTP/SAVPF 111", "c=IN IP4 0.0.0.0", "a=mid:0", "a=sendrecv",
  "a=setup:active", "a=rtpmap:111 opus/48000/2",
  "a=extmap:1 urn:ietf:params:rtp-hdrext:ssrc-audio-level",
  "m=video 0 UDP/TLS/RTP/SAVPF 96", "a=mid:1", "a=rtpmap:96 VP8/90000",
].join("\r\n") + "\r\n";

const holdAnswer = sdpAnswer.replace("a=sendrecv", "a=sendonly");

function sip(lines, body = "") {
  const head = lines.join("\r\n");
  return head + "\r\nContent-Length: " + body.length + "\r\n\r\n" + body;
}

const baseHdr = (extra) => [
  ...extra,
  "Via: SIP/2.0/UDP pc:5060",
  "Call-ID: call-abc@pc",
];

const invite = sip([
  "INVITE sip:bob@biloxi.com SIP/2.0",
  ...baseHdr(["From: Alice <sip:alice@a.com>;tag=alice1", "To: Bob <sip:bob@biloxi.com>"]),
  "CSeq: 1 INVITE", "Content-Type: application/sdp",
], sdpOffer);

const ringing = sip([
  "SIP/2.0 180 Ringing",
  ...baseHdr(["From: Alice <sip:alice@a.com>;tag=alice1", "To: Bob <sip:bob@biloxi.com>;tag=bob1"]),
  "CSeq: 1 INVITE",
]);

const ok200 = sip([
  "SIP/2.0 200 OK",
  ...baseHdr(["From: Alice <sip:alice@a.com>;tag=alice1", "To: Bob <sip:bob@biloxi.com>;tag=bob1"]),
  "CSeq: 1 INVITE", "Content-Type: application/sdp",
], sdpAnswer);

const ack = sip([
  "ACK sip:bob@biloxi.com SIP/2.0",
  ...baseHdr(["From: Alice <sip:alice@a.com>;tag=alice1", "To: Bob <sip:bob@biloxi.com>;tag=bob1"]),
  "CSeq: 1 ACK",
]);

const bye = sip([
  "BYE sip:alice@a.com SIP/2.0",
  ...baseHdr(["From: Bob <sip:bob@biloxi.com>;tag=bob1", "To: Alice <sip:alice@a.com>;tag=alice1"]),
  "CSeq: 2 BYE",
]);

// ─── SDP parsing ─────────────────────────────────────────────────────────────
test("parseSdp: codecs, extensions, bundle, dtls", () => {
  const p = parseSdp(sdpOffer);
  assert.deepEqual(p.bundle, ["0", "1"]);
  assert.equal(p.media.length, 2);
  const audio = p.media[0];
  assert.equal(audio.type, "audio");
  const opus = audio.codecs.find((c) => c.codec === "opus");
  assert.equal(opus.pt, 111);
  assert.equal(opus.rate, 48000);
  assert.equal(opus.fmtp, "minptime=10;useinbandfec=1");
  // static PT 0 (PCMU) recognized
  assert.ok(audio.codecs.find((c) => c.codec === "PCMU" && c.pt === 0));
  assert.equal(audio.extensions[0].uri, "urn:ietf:params:rtp-hdrext:ssrc-audio-level");
  assert.equal(audio.dtls.setup, "actpass");
  assert.equal(audio.ice.ufrag, "aaaa");
});

// ─── SIP message parsing ──────────────────────────────────────────────────────
test("parseSipMessage: request + response + tags + sdp", () => {
  const req = parseSipMessage(invite);
  assert.equal(req.kind, "request");
  assert.equal(req.method, "INVITE");
  assert.equal(req.callId, "call-abc@pc");
  assert.equal(req.fromTag, "alice1");
  assert.equal(req.toTag, null);
  assert.equal(req.cseq.method, "INVITE");
  assert.ok(req.sdp);

  const res = parseSipMessage(ok200);
  assert.equal(res.kind, "response");
  assert.equal(res.statusCode, 200);
  assert.equal(res.toTag, "bob1");
});

test("parseSipMessage: compact header forms", () => {
  const msg = "OPTIONS sip:x SIP/2.0\r\ni: c1@h\r\nf: <sip:a@h>;tag=t1\r\nt: <sip:b@h>\r\nl: 0\r\n\r\n";
  const p = parseSipMessage(msg);
  assert.equal(p.callId, "c1@h");
  assert.equal(p.fromTag, "t1");
});

// ─── full SIP dialog ──────────────────────────────────────────────────────────
test("SIP dialog: INVITE→180→200→ACK→BYE, FSM + negotiation", () => {
  resetDialogs();
  addMessage({ rawText: invite });
  addMessage({ rawText: ringing });
  const r = addMessage({ rawText: ok200 });
  const id = r.dialogId;
  assert.equal(r.state, "Confirmed");
  addMessage({ rawText: ack });
  const last = addMessage({ rawText: bye });
  assert.equal(last.state, "Terminated");

  const st = getState(id);
  assert.equal(st.steps, 5);
  const audio = st.negotiated.media.find((m) => m.type === "audio");
  assert.equal(audio.agreedCodecs[0].codec, "opus");
  assert.equal(audio.agreedExtensions[0].uri, "urn:ietf:params:rtp-hdrext:ssrc-audio-level");
  // rejected video m-line yields no agreed codecs
  const video = st.negotiated.media.find((m) => m.type === "video");
  assert.equal(video.rejected, true);
  assert.deepEqual(video.agreedCodecs, []);

  const tl = getTimeline(id);
  assert.equal(tl.messages.length, 5);
  assert.equal(tl.messages[0].label, "INVITE (offer)");
  assert.equal(tl.messages[2].label, "200 OK (answer)");
  assert.ok(tl.mermaid.startsWith("sequenceDiagram"));

  const cd = getCodecs(id);
  assert.ok(cd.perStep.length >= 2); // offer + answer carry media
});

test("SIP dialog: hold via re-INVITE sendonly", () => {
  resetDialogs();
  addMessage({ rawText: invite });
  addMessage({ rawText: ok200 });
  // re-INVITE offer with hold
  const reInvite = sip([
    "INVITE sip:bob@biloxi.com SIP/2.0",
    ...baseHdr(["From: Alice <sip:alice@a.com>;tag=alice1", "To: Bob <sip:bob@biloxi.com>;tag=bob1"]),
    "CSeq: 2 INVITE", "Content-Type: application/sdp",
  ], sdpOffer.replace("a=sendrecv", "a=sendonly"));
  const reOk = sip([
    "SIP/2.0 200 OK",
    ...baseHdr(["From: Alice <sip:alice@a.com>;tag=alice1", "To: Bob <sip:bob@biloxi.com>;tag=bob1"]),
    "CSeq: 2 INVITE", "Content-Type: application/sdp",
  ], holdAnswer);
  addMessage({ rawText: reInvite });
  const r = addMessage({ rawText: reOk });
  const st = getState(r.dialogId);
  assert.equal(st.state, "Confirmed"); // re-INVITE keeps Confirmed
  assert.equal(st.negotiated.hold, true);
});

test("SIP forking: one Call-ID, two To-tags → fork recorded", () => {
  resetDialogs();
  addMessage({ rawText: invite }); // early dialog, no To-tag
  // first 180 from UAS B
  addMessage({ rawText: ringing });
  // second 180 from a different UAS (To-tag bob2) — a fork
  const ringing2 = sip([
    "SIP/2.0 180 Ringing",
    ...baseHdr(["From: Alice <sip:alice@a.com>;tag=alice1", "To: Bob <sip:bob@biloxi.com>;tag=bob2"]),
    "CSeq: 1 INVITE",
  ]);
  addMessage({ rawText: ringing2 });
  const dialogs = listDialogs();
  // at least one dialog should record a fork branch
  const totalForks = dialogs.reduce((n, d) => n + d.forks, 0);
  assert.ok(totalForks >= 1, "expected a fork branch to be recorded");
});

// ─── pcap ─────────────────────────────────────────────────────────────────────
function udpFrame(payload, { sport = 5060, dport = 5060 } = {}) {
  const data = Buffer.from(payload, "utf8");
  const eth = Buffer.alloc(14);
  eth.writeUInt16BE(0x0800, 12); // IPv4
  const ip = Buffer.alloc(20);
  ip[0] = 0x45;
  ip.writeUInt16BE(20 + 8 + data.length, 2);
  ip[8] = 64; // ttl
  ip[9] = 17; // UDP
  ip.set([10, 0, 0, 1], 12); // src 10.0.0.1
  ip.set([10, 0, 0, 2], 16); // dst 10.0.0.2
  const udp = Buffer.alloc(8);
  udp.writeUInt16BE(sport, 0);
  udp.writeUInt16BE(dport, 2);
  udp.writeUInt16BE(8 + data.length, 4);
  return Buffer.concat([eth, ip, udp, data]);
}

function buildClassicPcap(frames) {
  const gh = Buffer.alloc(24);
  gh.writeUInt32LE(0xa1b2c3d4, 0); // magic (LE file)
  gh.writeUInt16LE(2, 4);
  gh.writeUInt16LE(4, 6);
  gh.writeUInt32LE(65535, 16); // snaplen
  gh.writeUInt32LE(1, 20); // linktype EN10MB
  const recs = frames.map((f, i) => {
    const rh = Buffer.alloc(16);
    rh.writeUInt32LE(1000 + i, 0); // ts_sec
    rh.writeUInt32LE(i, 4); // ts_usec
    rh.writeUInt32LE(f.length, 8);
    rh.writeUInt32LE(f.length, 12);
    return Buffer.concat([rh, f]);
  });
  return Buffer.concat([gh, ...recs]);
}

function buildPcapng(frames) {
  // SHB
  const shb = Buffer.alloc(28);
  shb.writeUInt32LE(0x0a0d0d0a, 0);
  shb.writeUInt32LE(28, 4);
  shb.writeUInt32LE(0x1a2b3c4d, 8); // byte-order magic
  shb.writeUInt16LE(1, 12); // major
  shb.writeUInt16LE(0, 14); // minor
  shb.writeUInt32LE(0xffffffff, 16); // section length lo (-1)
  shb.writeUInt32LE(0xffffffff, 20);
  shb.writeUInt32LE(28, 24); // trailing total_length
  // IDB
  const idb = Buffer.alloc(20);
  idb.writeUInt32LE(0x00000001, 0);
  idb.writeUInt32LE(20, 4);
  idb.writeUInt16LE(1, 8); // linktype EN10MB
  idb.writeUInt32LE(65535, 12); // snaplen
  idb.writeUInt32LE(20, 16);
  const epbs = frames.map((f) => {
    const pad = (4 - (f.length % 4)) % 4;
    const total = 28 + f.length + pad + 4;
    const b = Buffer.alloc(total);
    b.writeUInt32LE(0x00000006, 0);
    b.writeUInt32LE(total, 4);
    b.writeUInt32LE(0, 8); // interface_id
    b.writeUInt32LE(0, 12); // ts_high
    b.writeUInt32LE(1000, 16); // ts_low
    b.writeUInt32LE(f.length, 20); // cap_len
    b.writeUInt32LE(f.length, 24); // orig_len
    f.copy(b, 28);
    b.writeUInt32LE(total, total - 4);
    return b;
  });
  return Buffer.concat([shb, idb, ...epbs]);
}

test("parsePcap: classic pcap UDP/5060 → SIP messages", () => {
  const buf = buildClassicPcap([udpFrame(invite), udpFrame(ok200)]);
  const packets = parsePcap(buf);
  assert.equal(packets.length, 2);
  assert.equal(packets[0].proto, "udp");
  const msgs = reassembleSip(packets, { sipPort: 5060 });
  assert.equal(msgs.length, 2);
  assert.ok(msgs[0].raw.startsWith("INVITE"));
});

test("parsePcap: pcapng UDP/5060 → SIP dialog", () => {
  resetDialogs();
  const buf = buildPcapng([udpFrame(invite), udpFrame(ringing), udpFrame(ok200)]);
  const packets = parsePcap(buf);
  assert.equal(packets.length, 3);
  const r = addMessage({ pcapBuffer: buf });
  assert.equal(r.added, 3);
  assert.equal(r.state, "Confirmed");
});

test("parsePcap: bad magic throws", () => {
  assert.throws(() => parsePcap(Buffer.from([0x00, 0x11, 0x22, 0x33, 0x44, 0x55])), /unknown magic/);
});

// ─── offer/answer through a proxy (hop-aware) ─────────────────────────────────
test("sip_dialog: proxied call pairs each answer with the offer on its own hop", () => {
  resetDialogs();
  const r = addMessage({ pcapBuffer: proxyCallPcap() });
  assert.equal(r.added, 13);
  const codecs = getCodecs(r.dialogId);
  // INVITE A→P, INVITE P→B are both offers; both 200 OKs are answers
  assert.deepEqual(codecs.perStep.map((p) => `${p.step}:${p.sdpRole}`), ["0:offer", "2:offer", "5:answer", "6:answer"]);
  const st = getState(r.dialogId);
  // the last answer (200 OK P→A, step 6) answers Alice's INVITE (step 0), not the forwarded one
  assert.equal(st.negotiated.offerStep, 0);
  assert.equal(st.negotiated.answerStep, 6);
  assert.deepEqual(st.negotiated.media[0].agreedCodecs.map((c) => c.codec), ["PCMU", "telephone-event"]);
  assert.equal(st.state, "Terminated");
  const tl = getTimeline(r.dialogId);
  assert.equal(tl.messages[2].hop.dst, "203.0.113.20:5060");
});

test("sip_dialog: late offer (200 OK) is answered by ACK", () => {
  resetDialogs();
  const h = (cseq, to = "To: <sip:b@x>") => ["Via: SIP/2.0/UDP 10.0.0.1;branch=z9hG4bKlo", "From: <sip:a@x>;tag=fa", to, "Call-ID: late@x", `CSeq: ${cseq}`];
  const sdp = (pts) => `v=0\r\no=- 1 1 IN IP4 10.0.0.1\r\ns=-\r\nc=IN IP4 10.0.0.1\r\nt=0 0\r\nm=audio 4000 RTP/AVP ${pts.join(" ")}\r\n`;
  addMessage({ rawText: sipMsg("INVITE sip:b@x SIP/2.0", h("1 INVITE")) });
  addMessage({ rawText: sipMsg("SIP/2.0 200 OK", h("1 INVITE", "To: <sip:b@x>;tag=tb"), sdp([0, 8])) });
  const r = addMessage({ rawText: sipMsg("ACK sip:b@x SIP/2.0", h("1 ACK", "To: <sip:b@x>;tag=tb"), sdp([8])) });
  const roles = getCodecs(r.dialogId).perStep.map((p) => p.sdpRole);
  assert.deepEqual(roles, ["offer", "answer"]);
  const st = getState(r.dialogId);
  assert.equal(st.negotiated.offerStep, 1);
  assert.deepEqual(st.negotiated.media[0].agreedCodecs.map((c) => c.codec), ["PCMA"]);
});

test("sip_dialog: a retransmitted INVITE does not change state or offer/answer", () => {
  resetDialogs();
  const inv = sipMsg("INVITE sip:b@x SIP/2.0", ["Via: SIP/2.0/UDP 10.0.0.1;branch=z9hG4bKrt", "From: <sip:a@x>;tag=fr", "To: <sip:b@x>", "Call-ID: rt@x", "CSeq: 1 INVITE"],
    "v=0\r\no=- 1 1 IN IP4 10.0.0.1\r\ns=-\r\nc=IN IP4 10.0.0.1\r\nt=0 0\r\nm=audio 4000 RTP/AVP 0\r\n");
  const pcap = classicPcap([
    { ts: 0, frame: udp({ src: "10.0.0.1", dst: "10.0.0.2", payload: inv }) },
    { ts: 500, frame: udp({ src: "10.0.0.1", dst: "10.0.0.2", payload: inv }) },
  ]);
  const r = addMessage({ pcapBuffer: pcap });
  const tl = getTimeline(r.dialogId);
  assert.equal(tl.messages.length, 2);
  assert.match(tl.messages[1].label, /\[retransmission\]/);
  assert.deepEqual(getCodecs(r.dialogId).perStep.map((p) => p.sdpRole), ["offer", null]);
  assert.equal(getState(r.dialogId).state, "Trying");
});
